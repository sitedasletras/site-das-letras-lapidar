function continuarFluxo(){ const azul = localStorage.getItem("modulo_editorial")==="sim"; const amarelo = localStorage.getItem("modulo_audiovisual")==="sim"; const verde = localStorage.getItem("modulo_musica")==="sim"; if(azul){ window.location.href="/modulos_livro.html"; return; } if(amarelo){ window.location.href="/modulos_audiovisual.html"; return; } if(verde){ window.location.href="/modulos_musica.html"; return; } alert("Escolha pelo menos um tipo de produção antes de continuar."); }
